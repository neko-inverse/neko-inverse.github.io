<h3>效果实现（JS）</h3>
<p>具体说明都写在代码注释里了，这里直接上代码</p>
<pre><code>// index.js
/**
 * 解析歌词字符串
 * 得到一个歌词对象数组
 * @returns [{
 *    time: 开始时间
      words: 歌词内容
 * },]
 */
function parseLrc() {
  let lines = lrc.split(&#039;\n&#039;)
  let result = []
  lines.map(line =&gt; {
    let [timeStr, content] = line.split(&#039;]&#039;)
    timeStr = timeStr.substring(1)
    let obj = {
      time: parseTime(timeStr),
      words: content
    }
    result.push(obj)
  })
  return result
}

/**
 * 将时间字符串转换为数字（秒）
 * @param {string} timeStr 时间字符串
 * @returns {number} 时间秒数
 */
function parseTime(timeStr) {
  let [min, sec] = timeStr.split(&#039;:&#039;)
  return +min * 60 + +sec // 比后面这个传统写法高级点 parseInt(min) * 60 + parseInt(sec)
}

let lrcData = parseLrc()

// 获取需要的 dom
let doms = {
  audio: document.querySelector(&#039;audio&#039;),
  container: document.querySelector(&#039;.container&#039;),
  ul: document.querySelector(&#039;.container ul&#039;),
}

/**
 * 计算出，在当前播放器播放到第几秒的情况下
 * lrcData数组中，应该高亮显示的歌词下标
 * 如果没有任何一句歌词需要显示，则得到-1
 * @returns {number} 当前播放时间对应的歌词索引
 */
function findIndex() {
  let curTime = doms.audio.currentTime
  for (let i = 0; i &lt; lrcData.length; i++) {
    if (curTime &lt; lrcData[i].time) {
      return i - 1
    }
  }
  // 找遍了都没找到（说明播放到最后一句）返回最后一句的索引
  return lrcData.length - 1
}

/* 界面 */

/**
 * 创建歌词元素 li
 */
function createLrcElements() {
  // 创建文档片段
  let frag = document.createDocumentFragment()
  lrcData.map(item =&gt; {
    let li = document.createElement(&#039;li&#039;)
    li.textContent = item.words
    frag.appendChild(li)
  })
  doms.ul.appendChild(frag)
}

createLrcElements()

// 容器高度
let containerHeight = doms.container.clientHeight
// 歌词元素高度
let liHeight = doms.ul.children[0].clientHeight
// 最大偏移量 = ul 高度 - 容器高度
let maxOffset = doms.ul.clientHeight - containerHeight

/**
 * 设置 ul 元素的偏移量
 */
function setOffset() {
  let index = findIndex()
  // 偏移量 = 歌词元素高度 * 索引第几个 + 歌词元素高度的一半 - 容器高度的一半
  let offset = liHeight * index + liHeight / 2 - containerHeight / 2
  if(offset &lt; 0) offset = 0
  if(offset &gt; maxOffset) offset = maxOffset
  doms.ul.style.transform = `translateY(-${offset}px)`
  // 去掉之前的 li 的 active 样式
  let li = document.querySelector(&#039;.active&#039;)
  li?.classList.remove(&#039;active&#039;)
  // 设置当前的 li 的 active 样式
  li = doms.ul.children[index]
  li?.classList.add(&#039;active&#039;)
}

// 播放时间改变调用 setOffset 重新设置偏移量
doms.audio?.addEventListener(&#039;timeupdate&#039;, setOffset)</code></pre>
