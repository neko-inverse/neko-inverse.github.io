<h3>话不多说，代码展示</h3>
<p><strong>老规矩，解释在注释里</strong></p>
<pre><code class="language-html">&lt;!DOCTYPE html&gt;
&lt;html lang=&quot;en&quot;&gt;
&lt;head&gt;
  &lt;meta charset=&quot;UTF-8&quot;&gt;
  &lt;meta name=&quot;viewport&quot; content=&quot;width=device-width, initial-scale=1.0&quot;&gt;
  &lt;title&gt;randomColor&lt;/title&gt;
&lt;/head&gt;
&lt;body&gt;
  &lt;script&gt;
    /* 定义 randomColor 函数，先生成一个 0 到 1 的随机数，然后设置随机数范围为 0 到 0xffffff，
      因为生成的数值有小数，所以使用floor 函数取整，最后使用 toString(16) 将随机数转换为十六进制字符串 */
    let randomColor = () =&gt; `#${Math.floor(Math.random() * 0xffffff).toString(16)}`
    // 打印一下看看随机生成的十六进制颜色值
    console.log(randomColor())
    // 给整个 body 设置一个随机颜色背景看看效果，没刷新一次背景色会随机改变
    document.body.style.backgroundColor = randomColor()
  &lt;/script&gt;
&lt;/body&gt;
&lt;/html&gt;</code></pre>
