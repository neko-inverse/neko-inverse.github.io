      background-size: 0px 2px;
      transition: background-size 1300ms;
    }
    .title span:hover {
      /* 鼠标移入时，背景图宽度变为100% */
      background-size: 100% 2px;
      /* 背景图横向位置从左起，进入时从左开始，退出时样式失效，为上面的靠右起 */
